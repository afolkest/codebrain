# codebrain CLI cheatsheet

## Setup / sync

python3 -m codebrain collect --pool ~/codebrain-pool
# Copy local raw session/history files into the syncable pool.

python3 -m codebrain collect --install-launchd --pool ~/codebrain-pool --interval 300
# Keep the pool fresh every five minutes on this Mac.

python3 -m codebrain ingest-pool --machine macbook --source pi
# Debug/repair: explicitly ingest one synced remote pool subtree.
# Normal read commands refresh remote pool subtrees on demand.

## Daily archaeology loop

python3 -m codebrain recent
# Sessions sorted by latest user message.

python3 -m codebrain recent --limit 10
# Top 10 recent intent-bearing sessions.

python3 -m codebrain recent --cwd codebrain
# Recent sessions whose cwd contains "codebrain".

python3 -m codebrain recent --cwd example-project --min-chars 80 --max-chars 700
# Recent example-project sessions using medium-sized user messages as signal.

python3 -m codebrain recent --source pi --since 7d
# Recent pi sessions from the last 7 days.

python3 -m codebrain recent --json --limit 5
# Same as recent, but scriptable JSON.

python3 -m codebrain userlog
# Newest user messages across all sessions.

python3 -m codebrain userlog --limit 20 --min-chars 80 --max-chars 700
# Recent medium-sized user messages; often highest signal.

python3 -m codebrain userlog --cwd getrandomness --limit 20
# Recent user messages in getrandomness sessions.

python3 -m codebrain userlog --query "prefer" --limit 20
# Recent user messages containing "prefer".

python3 -m codebrain userlog --since 30d --source codex
# User messages in Codex sessions from the last 30 days.

python3 -m codebrain userlog --json --limit 10
# Userlog as JSON for scripts/agents.

python3 -m codebrain search "sqlite refresh"
# FTS search over live, authored event text.

python3 -m codebrain search "ExamplePriority" --actor user --before 2026-06-11
# Search user messages before a date; useful for decision archaeology without current-session echoes.

python3 -m codebrain search "ExamplePriority" --exclude-session <session_id>
# Explicitly remove a noisy/current session from results.

python3 -m codebrain search "ExamplePriority" --only-session <session_id>
# Deliberately search one long/current session, including inherited live context for pre-compact recovery.

python3 -m codebrain search "ExamplePriority" --exclude-recent 1h --json
# Scriptable search, excluding very recent self-referential results.

python3 -m codebrain search "ExamplePriority" --actor user --around 2
# Search and inline nearby user-centered turns, with tools hidden by default.

python3 -m codebrain turns <session_id>
# Show a session as user-centered turns.

python3 -m codebrain turns <session_id> --around-seq <seq>
# Expand around a seq from recent/userlog/search.

python3 -m codebrain turns <session_id> --around-seq <seq> --context-turns 1
# One turn before and after the target seq.

python3 -m codebrain turns <session_id> --agent-chars 120
# Shorter assistant previews.

python3 -m codebrain turns <session_id> --show-tools --tool-chars 80
# Include truncated tool calls/results.

python3 -m codebrain turns <session_id> --all
# Include rolled-back turns/events too.

python3 -m codebrain lineage <session_id>
# Show factual parent/root/children/sibling lineage for a session.

python3 -m codebrain lineage <session_id> --json
# Lineage as JSON for scripts/agents.

## Artifact handoff

python3 -m codebrain refs <session_id>
# Files, commands, and commit hashes referenced by a session.

python3 -m codebrain refs <session_id> --around-seq <seq> --context-turns 1
# References only near a specific turn/window.

python3 -m codebrain refs <session_id> --json
# References as JSON for scripts/agents.

python3 -m codebrain touched docs/wip/pipeline-redesign.md
# Sessions/events with that structured file ref; relative path queries also match absolute refs.

python3 -m codebrain touched pipeline-redesign.md --basename
# Match structured file refs by basename.

python3 -m codebrain touched docs/wip/ --prefix --cwd example-project
# Match structured refs under a directory/path prefix.

python3 -m codebrain touched docs/wip/pipeline-redesign.md --json
# File-first archaeology as JSON for scripts/agents.

## Escape hatches

python3 -m codebrain show <session_id>
# Raw live transcript view. Usually start with turns instead.

python3 -m codebrain show <session_id> --all
# Raw transcript including rolled-back events.

python3 -m codebrain list
# Session metadata by start time. Usually recent is better for intent archaeology.

python3 -m codebrain grep "some literal string"
# Grep local live raw logs directly.

python3 -m codebrain grep "some literal string" ~/codebrain-pool/raw
# Grep synced/archive raw logs directly.

python3 -m codebrain schema
# Print SQLite schema.

sqlite3 ~/.codebrain/codebrain.db
# Open the database directly for SQL.
